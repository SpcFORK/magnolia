_registry := []



fn _registerShader(shader) {

	_registry <- _registry << shader

	?

}



fn unregisterShader(shader) {

	_registry <- _registry |> with filter() fn(s) s != shader

}



fn clearAll() with each(_registry) fn(s) {

	if s != ? -> {

		reset(s)

		s._frameCount <- 0

	}

}



fn destroyAll() {

	with each(_registry) fn(s) {

		if s != ? -> {

			reset(s)

			s._frameCount <- 0

		}

	}

	_registry <- []

}



fn registeredCount() len(_registry)

cs Shader(fragment?, opts?) {

	fragment := fragment?

	draw := opts?.draw

	glslSource := opts?.glslSource

	resolution := opts?.resolution |> default(4)

	uniforms := opts?.uniforms |> default({})

	_startNano := nanotime()

	_pausedAt := ?

	_pauseAccum := 0

	_frameStart := 0

	_dt := 0

	_frameCount := 0

	_glslSubmitted := false

	_registered := _registerShader(Self)

}



fn elapsed(shader) {

	now := nanotime()

	active := if shader._pausedAt != ? {

		true -> shader._pausedAt - shader._startNano - shader._pauseAccum

		_ -> now - shader._startNano - shader._pauseAccum

	}

	float(active) / 1000000000.0

}



fn pause(shader) if shader._pausedAt = ? -> {

	shader._pausedAt <- nanotime()

}



fn resume(shader) if shader._pausedAt != ? -> {

	shader._pauseAccum <- shader._pauseAccum + (nanotime() - shader._pausedAt)

	shader._pausedAt <- ?

}



fn reset(shader) if shader != ? -> {

	shader._startNano <- nanotime()

	shader._pauseAccum <- 0

	shader._pausedAt <- ?

}



fn setUniform(shader, key, value) shader.uniforms.(key) <- value

fn getUniform(shader, key) shader.uniforms.(key)

fn setResolution(shader, res) shader.resolution <- res



fn beginFrame(shader) {

	shader._frameStart <- nanotime()

}



fn endFrame(shader) {

	now := nanotime()

	shader._dt <- float(now - shader._frameStart) / 1000000000.0

	shader._frameCount <- shader._frameCount + 1

	shader._dt

}



fn dt(shader) shader._dt

fn isRunning(shader) shader._pausedAt = ?

fn frameCount(shader) shader._frameCount

// @ Render Dispatch

// =========================================================================



fn render(window, shader, x, y, w, h) {

	t := elapsed(shader)



	// GPU path: if shader has a glslSource field and window is web backend,

	// use WebGL rendering and skip all CPU work (initRain, gradient bands, etc.)

	if shader.glslSource != ? & window.backend = :web -> {

		if shader._glslSubmitted != true -> {

			codegen.submitWebGL(window, shader.glslSource)

			shader._glslSubmitted <- true

		}

		codegen.drawWebGL(window)

	}



	// CPU path: custom draw callback or generic shader renderer

	if shader.glslSource = ? | window.backend != :web -> {

		if shader.draw != ? -> shader.draw(window, x, y, w, h, t, shader.uniforms)

		if shader.draw = ? -> renderShaderLines(window, shader, x, y, w, h)

	}

}

	shader := shader

	x := x

	y := y

	w := w

	h := h

}



fn composePasses(window, passes) {

	with each(passes) fn(pass) {

		render(window, pass.shader, pass.x, pass.y, pass.w, pass.h)

	}

}

fn renderShader(window, shader, x, y, w, h) {

	t := elapsed(shader)

	res := shader.resolution

	u := shader.uniforms



	fn _row(ry) if ry >= h {

		true -> ?

		_ -> {

			fn _col(rx) if rx >= w {

				true -> ?

				_ -> {

					c := shader.fragment(rx, ry, w, h, t, u)

					if c != ? -> _draw.fillRect(window, x + rx, y + ry, res, res, c)

					_col(rx + res)

				}

			}

			_col(0)

			_row(ry + res)

		}

	}

	_row(0)

}



fn renderShaderLines(window, shader, x, y, w, h) {

	t := elapsed(shader)

	res := shader.resolution

	u := shader.uniforms



	fn _row(ry) if ry >= h {

		true -> ?

		_ -> {

			c := shader.fragment(int(w / 2), ry, w, h, t, u)

			if c != ? -> _draw.drawLine(window, x, y + ry, x + w, y + ry, c)

			_row(ry + res)

		}

	}

	_row(0)

}



fn renderGradientBands(window, gradientFn, x, y, w, h, time, bands?) {

	n := if bands? { ? -> 16, _ -> bands? }

	bandH := int(h / n)

	if bandH < 1 -> bandH <- 1

	// Merge adjacent bands with same color into single fillRect calls

	fn _band(i, mergeY, mergeH, mergeC) if i >= n {

		true -> if mergeC != ? -> _draw.fillRect(window, x, y + mergeY, w, mergeH, mergeC)

		_ -> {

			by := i * bandH

			bh := if i = n - 1 { true -> h - by, _ -> bandH }

			c := gradientFn(by + int(bh / 2), h, time)

			if c = mergeC {

				true -> _band(i + 1, mergeY, mergeH + bh, mergeC)

				_ -> {

					if mergeC != ? -> _draw.fillRect(window, x, y + mergeY, w, mergeH, mergeC)

					_band(i + 1, by, bh, c)

				}

			}

		}

	}

	_band(0, 0, 0, ?)

}



fn renderGradient(window, gradientFn, x, y, w, h, time) renderGradientBands(window, gradientFn, x, y, w, h, time, 16)



fn renderHorizontalBands(window, gradientFn, x, y, w, h, time, bands?) {

	n := if bands? { ? -> 16, _ -> bands? }

	bandW := int(w / n)

	if bandW < 1 -> bandW <- 1

	fn _band(i) if i >= n {

		true -> ?

		_ -> {

			bx := i * bandW

			bw := if i = n - 1 { true -> w - bx, _ -> bandW }

			c := gradientFn(bx + int(bw / 2), w, time)

			if c != ? -> _draw.fillRect(window, x + bx, y, bw, h, c)

			_band(i + 1)

		}

	}

	_band(0)

}



fn renderColumns(window, columns, ox, oy, h) {

	with each(columns) fn(col_) {

		tailY := col_.headY - col_.length

		if tailY < 0 -> {

			_draw.drawLine(window, ox + col_.x, oy + 0, ox + col_.x, oy + col_.headY, col_.color)

			_draw.drawLine(window, ox + col_.x, oy + h + tailY, ox + col_.x, oy + h, col_.color)

		}

		if tailY >= 0 -> {

			_draw.drawLine(window, ox + col_.x, oy + tailY, ox + col_.x, oy + col_.headY, col_.color)

		}

	}

}



fn updateColumns(columns, h, t, rate?) {

	fh := float(h)

	r := if rate? { ? -> 40.0, _ -> rate? }

	invH := r / fh

	with each(columns) fn(col_) {

		col_.headY <- int(fract(col_.phase + t * col_.speed * invH) * fh)

	}

}

fn createBuffer(w, h) { width: w, height: h, data: {} }



fn clearBuffer(buf, color?) {

	c := if color? { ? -> 0, _ -> color? }

	total := buf.width * buf.height

	fn _clear(i) if i < total -> {

		buf.data.(i) <- c

		_clear(i + 1)

	}

	_clear(0)

}



fn setPixel(buf, x, y, color) if {

	x < 0 -> ?

	y < 0 -> ?

	x >= buf.width -> ?

	y >= buf.height -> ?

	_ -> buf.data.(y * buf.width + x) <- color

}



fn getPixel(buf, x, y) if {

	x < 0 -> 0

	y < 0 -> 0

	x >= buf.width -> 0

	y >= buf.height -> 0

	_ -> {

		v := buf.data.(y * buf.width + x)

		if v { ? -> 0, _ -> v }

	}

}



fn renderBuffer(window, buf, ox, oy) {

	fn _row(ry) if ry < buf.height -> {

		fn _col(rx, runX, runW, runC) if rx >= buf.width {

			true -> {

				if runC != 0 & runW > 0 -> _draw.fillRect(window, ox + runX, oy + ry, runW, 1, runC)

				?

			}

			_ -> {

				c := getPixel(buf, rx, ry)

				if c = runC {

					true -> _col(rx + 1, runX, runW + 1, runC)

					_ -> {

						if runC != 0 & runW > 0 -> _draw.fillRect(window, ox + runX, oy + ry, runW, 1, runC)

						_col(rx + 1, rx, 1, c)

					}

				}

			}

		}

		_col(0, 0, 0, 0)

		_row(ry + 1)

	}

	_row(0)

}



fn renderShaderToBuffer(buf, shader) {

	t := elapsed(shader)

	w := buf.width

	h := buf.height

	u := shader.uniforms

	res := shader.resolution

	fn _row(ry) if ry < h -> {

		fn _col(rx) if rx < w -> {

			c := shader.fragment(rx, ry, w, h, t, u)

			if c != ? -> setPixel(buf, rx, ry, c)

			_col(rx + res)

		}

		_col(0)

		_row(ry + res)

	}

	_row(0)

}



// ─── Parallel shader rendering ──────────────────────────────────────────────

// Splits the framebuffer into horizontal bands and evaluates the fragment

// shader in parallel using worker goroutines.



fn renderShaderToBufferParallel(buf, shader, numWorkers) {

	t := elapsed(shader)

	w := buf.width

	h := buf.height

	u := shader.uniforms

	res := shader.resolution

	workers := if numWorkers { ? -> 4, _ -> numWorkers }



	// Split into horizontal bands

	bandH := int(h / workers) + 1

	bands := []

	startY := 0

	fn _makeBands() if startY < h -> {

		endY := startY + bandH

		if endY > h -> endY <- h

		bands << { startY: startY, endY: endY }

		startY <- endY

		_makeBands()

	}

	_makeBands()



	// Each worker computes pixels for its band into a local results list

	bandResults := threadLib.pmap(bands, fn(band) {

		results := []

		fn _row(ry) if ry < band.endY -> {

			fn _col(rx) if rx < w -> {

				c := shader.fragment(rx, ry, w, h, t, u)

				if c != ? -> results << { x: rx, y: ry, c: c }

				_col(rx + res)

			}

			_col(0)

			_row(ry + res)

		}

		_row(band.startY)

		results

	})



	// Merge results back into the buffer (serial, fast writes)

	with each(bandResults) fn(results) {

		with each(results) fn(px) setPixel(buf, px.x, px.y, px.c)

	}

}



// renderParallel evaluates a shader and draws directly to the window

// using parallel horizontal band computation.

fn renderParallel(window, shader, x, y, w, h, numWorkers) {

	t := elapsed(shader)

	u := shader.uniforms

	res := default(shader.resolution, 1)

	workers := if numWorkers { ? -> 4, _ -> numWorkers }



	bandH := int(h / workers) + 1

	bands := []

	startY := 0

	fn _mkB() if startY < h -> {

		endY := startY + bandH

		if endY > h -> endY <- h

		bands << { startY: startY, endY: endY }

		startY <- endY

		_mkB()

	}

	_mkB()



	// Compute pixel colors in parallel (pure math, no GDI)

	bandPixels := threadLib.pmap(bands, fn(band) {

		pixels := []

		fn _row(ry) if ry < band.endY -> {

			fn _col(rx) if rx < w -> {

				c := shader.fragment(rx, ry, w, h, t, u)

				if c != ? -> pixels << { x: x + rx, y: y + ry, c: c }

				_col(rx + res)

			}

			_col(0)

			_row(ry + res)

		}

		_row(band.startY)

		pixels

	})



	// Replay all pixels on the main thread (serial GDI)

	with each(bandPixels) fn(pixels) {

		with each(pixels) fn(px) _draw.fillRect(window, px.x, px.y, res, res, px.c)

	}

}
