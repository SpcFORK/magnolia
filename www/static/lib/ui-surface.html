// ---------------------------------------------------------------------------
// Contract fingerprint — bump when any code above changes meaning
// ---------------------------------------------------------------------------

// fingerprint returns a stable string identifying this surface revision.
// Used by drift tests and host ABI checks.
fn fingerprint() {
	'ui-surface/v' + string(SURFACE_VERSION) +
		'/ops=' + string(OP.debugStep) +
		'/props=' + string(PROP.rotate) +
		'/draw=' + string(DRAW_OP.clear)
}

// opName returns the string name for an op code, or ? if unknown.
fn opName(code) {
	ks := keys(OP)
	fn loop(i) if i < len(ks) {
		true -> {
			k := ks.(i)
			if OP.(k) = code {
				true -> k
				_ -> loop(i + 1)
			}
		}
		_ -> ?
	}
	loop(0)
}

// propName returns the string name for a prop id, or ?.
fn propName(code) {
	ks := keys(PROP)
	fn loop(i) if i < len(ks) {
		true -> {
			k := ks.(i)
			if PROP.(k) = code {
				true -> k
				_ -> loop(i + 1)
			}
		}
		_ -> ?
	}
	loop(0)
}

// knownOp? reports whether code is a defined surface op.
fn knownOp?(code) opName(code) != ?

// knownProp? reports whether code is a defined prop id.
fn knownProp?(code) propName(code) != ?

// packId builds a generation-tagged handle from slot + generation.
fn packId(slot, gen) (gen * 1048576) + (slot % 1048576)

// unpackSlot extracts the arena slot from a generation-tagged id.
fn unpackSlot(id) id % 1048576

// unpackGen extracts the generation from a generation-tagged id.
fn unpackGen(id) int(id / 1048576)

// abgr packs r,g,b,a (0..255) into a u32 ABGR color (low byte = R).
fn abgr(r, g, b, a) {
	aa := if a { ? -> 255, _ -> a }
	(aa % 256) * 16777216 + (b % 256) * 65536 + (g % 256) * 256 + (r % 256)
}

// rgb is abgr with full alpha.
fn rgb(r, g, b) abgr(r, g, b, 255)
