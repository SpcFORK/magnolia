// linux-libc provides Linux process, fs, and memory helpers.

sys := import('sys')
{
	readU32: readU32
} := import('writes')

{
	PROT_READ: PROT_READ
	PROT_WRITE: PROT_WRITE
	MAP_PRIVATE: MAP_PRIVATE
	MAP_ANONYMOUS: MAP_ANONYMOUS
	RTLD_NOW: RTLD_NOW
	RTLD_LOCAL: RTLD_LOCAL
	_SC_PAGESIZE: _SC_PAGESIZE
} := import('linux-constants')

{
	isLinux?: isLinux?
	cstr: cstr
	_readCString: _readCString
	_platformError: _platformError
	libc: libc
	libdl: libdl
} := import('linux-core')

{
	loadLibrary: loadLibrary
} := import('linux-loader')

fn currentProcessId() if isLinux?() {
	true -> sys.valueOr(libc('getpid'), -1)
	_ -> -1
}

fn parentProcessId() if isLinux?() {
	true -> sys.valueOr(libc('getppid'), -1)
	_ -> -1
}

fn pageSize() if isLinux?() {
	true -> sys.valueOr(libc('sysconf', _SC_PAGESIZE), -1)
	_ -> -1
}

fn errno() if isLinux?() {
	true -> {
		loc := libc('__errno_location')
		if sys.ok?(loc) & loc.r1 > 0 {
			true -> readU32(loc.r1)
			_ -> -1
		}
	}
	_ -> -1
}

fn strerror(errorCode) if isLinux?() {
	true -> {
		msgPtr := libc('strerror', errorCode)
		if sys.ok?(msgPtr) & msgPtr.r1 > 0 {
			true -> _readCString(msgPtr.r1, 1024)
			_ -> ?
		}
	}
	_ -> ?
}

fn lastErrorMessage() if isLinux?() {
	true -> {
		err := errno()
		if err >= 0 {
			true -> strerror(err)
			_ -> ?
		}
	}
	_ -> ?
}

// Compatibility aliases for cross-platform interop callsites.
fn getLastError() errno()

fn formatMessage(errorCode) strerror(errorCode)

fn currentProcess() currentProcessId()

fn moduleHandle(name) if isLinux?() {
	true -> if name {
		? -> libdl('dlopen', 0, RTLD_NOW | RTLD_LOCAL)
		_ -> loadLibrary(name)
	}
	_ -> _platformError('moduleHandle')
}

fn imageBase() if isLinux?() {
	true -> sys.valueOr(moduleHandle(?), 0)
	_ -> 0
}

	true -> libc('open', cstr(path), flags, if mode {
		? -> 420

fn allocPages(size, prot) if isLinux?() {
	true -> mmap(
		?
		size
		if prot {
			? -> PROT_READ | PROT_WRITE
			_ -> prot
		}
		MAP_PRIVATE | MAP_ANONYMOUS
		-1
		0
	)
	_ -> _platformError('allocPages')
}

fn freePages(address, size) if isLinux?() {
	true -> munmap(address, size)
	_ -> _platformError('freePages')
}

fn protectPages(address, size, prot) if isLinux?() {
	true -> mprotect(address, size, prot)
	_ -> _platformError('protectPages')
}

fn virtualAlloc(baseAddress, size, allocationType, protection) if isLinux?() {
	true -> allocPages(size, protection)
	_ -> _platformError('virtualAlloc')
}

fn virtualFree(address, size, freeType) if isLinux?() {
	true -> freePages(address, size)
	_ -> _platformError('virtualFree')
}

fn virtualProtect(address, size, newProtect, oldProtectOutPtr) if isLinux?() {
	true -> protectPages(address, size, newProtect)
	_ -> _platformError('virtualProtect')
}

// Windows-only API compatibility stubs exposed for cross-platform symbol parity.
fn _compatNotImplemented(apiName) {
	{
		type: :error
		error: apiName + ' is not implemented on Linux compatibility layer'
		api: apiName
		platform: 'linux'
		supported: false
	}
}
