// Per-library proc address caches — avoids repeated loadDll/procAddress/sysproc

// round-trips on every call.  First call resolves the full chain and caches the

// integer address; subsequent calls go straight to syscall.

_loaderProcCache := {}



fn _cachedCallIn(library, symbol, args...) if _isWindowsPlatform {

	true -> {

		cacheKey := library + symbol

		cachedAddr := _loaderProcCache.(cacheKey)

		if cachedAddr {

			? -> {

				resolved := resolveInLoaded(library, symbol)

				if resolved.type = :ok {

					true -> {

						_loaderProcCache.(cacheKey) <- resolved.proc

						result := syscall(resolved.proc, args...)

						if result.type = :success -> result.type <- :ok

						result

					}

					_ -> resolved

				}

			}

			_ -> {

				result := syscall(cachedAddr, args...)

				if result.type = :success -> result.type <- :ok

				result

			}

		}

	}

	_ -> _platformError('callIn')

}



fn _normalizeHandleResult(result, apiName, library) if sys.ok?(result) & result.r1 > 0 {

	true -> {

		{

			type: :ok

			handle: result.r1

			library: library

		}

	}

	_ -> {

		{

			type: :error

			error: apiName + ' failed'

			library: library

			detail: result

		}

	}

}



fn loadDll(library) if isWindows?() {

	true -> if cached := _loadedLibraries.(library) {

		? -> {

			alreadyLoaded := moduleHandle(library)

			if sys.ok?(alreadyLoaded) & alreadyLoaded.r1 > 0 {

				true -> {

					_loadedLibraries.(library) := alreadyLoaded.r1

					{

						type: :ok

						handle: alreadyLoaded.r1

						library: library

					}

				}

				_ -> {

					loaded := loadLibrary(library)

					result := _normalizeHandleResult(loaded, 'LoadLibraryW', library)

					if result.type {

						:ok -> {

							_loadedLibraries.(library) := result.handle

							result

						}

						_ -> {

							systemPath := 'C:\\Windows\\System32\\' + library

							loadedSystem := loadLibrary(systemPath)

							systemResult := _normalizeHandleResult(loadedSystem, 'LoadLibraryW', systemPath)

							if systemResult.type {

								:ok -> {

									_loadedLibraries.(library) := systemResult.handle

									{

										type: :ok

										handle: systemResult.handle

										library: library

										path: systemPath

									}

								}

								_ -> {

									{

										type: :error

										error: 'LoadLibraryW failed'

										library: library

										detail: result.detail

										fallback: systemResult.detail

									}

								}

							}

						}

					}

				}

			}

		}

		_ -> {

			{

				type: :ok

				handle: cached

				library: library

			}

		}

	}

	_ -> _platformError('loadDll')

}



fn resolveInLoaded(library, symbol) if isWindows?() {

	true -> {

		loaded := loadDll(library)

		if loaded.type {

			:ok -> {

				proc := procAddress(loaded.handle, symbol)

				if sys.ok?(proc) & proc.r1 > 0 {

					true -> {

						{

							type: :ok

							proc: proc.r1

							handle: loaded.handle

							library: library

							symbol: symbol

						}

					}

					_ -> {

						{

							type: :error

							error: 'Could not resolve procedure'

							library: library

							symbol: symbol

							detail: proc

						}

					}

				}

			}

			_ -> loaded

		}

	}

	_ -> _platformError('resolveInLoaded')

}

fn apiSetDll(contract) {

	if contract {

		? -> ApiSetPrefix

		_ -> if len(contract) >= 4 {

			true -> if contract.(len(contract) - 4) = '.dll' {

				true -> contract

				_ -> ApiSetPrefix + contract + '.dll'

			}

			_ -> ApiSetPrefix + contract + '.dll'

		}

	}

}

fn _familyDll(prefix, suffix) if suffix {

	? -> prefix + '.dll'

	_ -> if len(suffix) >= 4 {

		true -> if suffix.(len(suffix) - 4) = '.dll' {

			true -> suffix

			_ -> prefix + suffix + '.dll'

		}

		_ -> prefix + suffix + '.dll'

	}

}
