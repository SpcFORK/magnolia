fn createWindow(title, width, height, options) {

	resolvedTitle := _default(title, 'Magnolia GUI Window')

	resolvedWidth := _default(width, 960)

	resolvedHeight := _default(height, 540)

	opts := _default(options, {})

	// Use the pre-registered default class name when no custom className

	// or classStyle is specified.  This avoids the ~500ms RegisterClassExW

	// call on the first window because the class is already registered at

	// module init time in windows-windowing.

	className := if opts.className {

		? -> if opts.classStyle {

			? -> if backend() {

				:windows -> windows.DefaultClassName

				_ -> 'MagnoliaGUIWindowClass_' + string(nanotime())

			}

			_ -> 'MagnoliaGUIWindowClass_' + string(nanotime())

		}

		_ -> opts.className

	}

	frameMs := _default(opts.frameMs, 16)

	updateOnDispatch := _default(opts.updateOnDispatch, false)



	created := if backend() {

		:windows -> guiNativeWin.createWindowState(resolvedTitle, resolvedWidth, resolvedHeight, opts, className, frameMs, updateOnDispatch)

		:linux -> guiNativeLinux.createWindowState(resolvedTitle, resolvedWidth, resolvedHeight, frameMs, updateOnDispatch)

		:web -> guiWeb.createWindowState(resolvedTitle, resolvedWidth, resolvedHeight, frameMs, updateOnDispatch, opts)

		_ -> _err('Unsupported runtime for GUI window middleware', ___runtime_sys_info())

	}



	if created.type {

		:ok -> {

			_ensureEventBus(created)

			// Auto-configure design resolution if specified in options.

			// designWidth/designHeight define the logical coordinate space;

			// the window is created at the requested physical size and the

			// back-buffer renders at the design resolution, then scales up.

			if opts.designWidth != ? & opts.designHeight != ? -> {

				guiResolution.setDesignResolution(created, opts.designWidth, opts.designHeight, {

					scaleMode: opts.scaleMode

					pixelPerfect: opts.pixelPerfect

				})

			}

			// Auto-init threading when threading options are specified.

			if _default(opts.threading, false) {

				true -> guiThread.initWindowThreading(created, {

					numWorkers: _default(opts.threadWorkers, 4)

					commandQueue: true

					workerPool: true

				})

			}

			created

		}

		_ -> created

	}

}



// Starts window creation in a background goroutine and returns a future immediately.

// Call awaitWindow(future) to block until the window is ready.

// This lets the caller do other setup work while the window is being created.

fn createWindowAsync(title, width, height, options) {

	resolvedTitle := _default(title, 'Magnolia GUI Window')

	resolvedWidth := _default(width, 960)

	resolvedHeight := _default(height, 540)

	opts := _default(options, {})

	className := if opts.className {

		? -> if opts.classStyle {

			? -> if backend() {

				:windows -> windows.DefaultClassName

				_ -> 'MagnoliaGUIWindowClass_' + string(nanotime())

			}

			_ -> 'MagnoliaGUIWindowClass_' + string(nanotime())

		}

		_ -> opts.className

	}

	frameMs := _default(opts.frameMs, 16)

	updateOnDispatch := _default(opts.updateOnDispatch, false)



	if backend() {

		:windows -> guiNativeWin.createWindowAsync(resolvedTitle, resolvedWidth, resolvedHeight, opts, className, frameMs, updateOnDispatch)

		_ -> {

			// Non-Windows backends: create synchronously and wrap in future.

			result := createWindow(title, width, height, options)

			{ type: :future, done: true, result: result }

		}

	}

}



// Blocks until an async window future completes, then returns the window state.

fn awaitWindow(future) if future.type {

	:future -> {

		result := if future.done {

			true -> future.result

			_ -> if backend() {

				:windows -> guiNativeWin.awaitWindow(future)

				_ -> future.result

			}

		}

		if result.type = :ok -> _ensureEventBus(result)

		result

	}

	_ -> future

}

	sx := _default(scaleX, 1)

	sy := _default(scaleY, sx)

	newWidth := int(float(_default(window.width, 1)) * float(sx))

	newHeight := int(float(_default(window.height, 1)) * float(sy))

	if newWidth < 1 -> newWidth <- 1

	if newHeight < 1 -> newHeight <- 1

	resize(window, newWidth, newHeight)

}

// getScreenCenter(window) -> { type: :ok, x, y } | error

// Coordinates that would center `window` on the primary screen.

fn getScreenCenter(window) {

	size := getPrimaryScreenSize(window)

	if size.type = :ok {

		true -> {

			coords := _screenPositionFor(size, _default(window.width, 0), _default(window.height, 0), :center, 0)

			{ type: :ok, x: coords.x, y: coords.y }

		}

		_ -> _err('getScreenCenter failed', size)

	}

}

		_ -> { type: :ok }

	}

	_ -> window

}



fn endFrame(window) if window.type {

	:ok -> if window.backend {

		:web -> guiWeb.endFrame(window)

		:windows -> {

			res := guiNativeWin.endFrame(window)

			if window._frameStart {

				? -> 0

				_ -> {

					now := nanotime()

					delta := now - window._frameStart

					window._fpsCount <- _default(window._fpsCount, 0) + 1

					window._fpsAccum <- _default(window._fpsAccum, 0) + delta

					if now - _default(window._fpsLastReport, 0) >= 1000000000 -> {

						fps := int(window._fpsCount * 1000000000 / window._fpsAccum)

						titleBase := _default(window._titleBase, window.title)

						window._titleBase <- titleBase

						setTitle(window, titleBase + ' - FPS: ' + string(fps))

						std.println('FPS: ' + string(fps))

						window._fpsCount <- 0

						window._fpsAccum <- 0

						window._fpsLastReport <- now

					}

				}

			}

			res

		}

		_ -> 0

	}

	_ -> 0

}

fn poll(window) if window.type {

	:ok -> if window.backend {

		:windows -> guiNativeWin.poll(window)

		:linux -> guiNativeLinux.poll(window)

		:web -> if window.closed {

			true -> { type: :closed }

			_ -> { type: :idle }

		}

		_ -> _err('Unknown window backend', window)

	}

	_ -> window

}

fn run(window, onEvent, onFrame) if window.type {

	:ok -> {

		if !_default(window._runStarted, false) -> {

			window._runStarted <- true

			_publish(window, :runStart, {

				type: :runStart

			})

			if _default(window.visible, false) {

				true -> _maybeRunFrame(window, onFrame, true)

			}

		}

		step := poll(window)

		if step.type {

			:dispatch -> {

				guiEvents._cacheDispatchContext(window)

				_markUrgentFrameIfResizeDispatch(window, step)

				_publish(window, :dispatch, step)

				if onEvent {

					? -> ?

					_ -> onEvent(window, step)

				}

				guiEvents._clearDispatchContext(window)

				// Render when frame is due. updateOnDispatch forces a frame on

				// every dispatch; otherwise only the timer-scheduled frame runs.

				_maybeRunFrame(window, onFrame, _default(window.updateOnDispatch, false))

				run(window, onEvent, onFrame)

			}

			:idle -> {

				if guiEvents.listenerCount(window, :idle) > 0 -> _publish(window, :idle, step)

				_sleepUntilNextFrame(window)

				_maybeRunFrame(window, onFrame, false)

				if window.backend = :web {

					true -> 0

					_ -> run(window, onEvent, onFrame)

				}

			}

			:closed -> {

				if window.backend = :windows {

					true -> close(window)

					_ -> {

						_publish(window, :closed, step)

						0

					}

				}

			}

			_ -> step

		}

	}

	_ -> window

}



fn close(window) if window.type {

	:ok -> if _default(window._nativeClosed, false) {

		true -> 0

		_ -> {

			_publish(window, :closing, {

				type: :closing

			})

			guiDraw.releaseResources(window)

			result := if window.backend {

				:windows -> guiNativeWin.close(window)

				:linux -> guiNativeLinux.close(window)

				:web -> {

					window.closed <- true

					window._nativeClosed <- true

					0

				}

				_ -> _err('Unknown window backend', window)

			}

			window._nativeClosed <- true

			_publish(window, :closed, {

				type: :closed

			})

			result

		}

	}

	_ -> 0

}
